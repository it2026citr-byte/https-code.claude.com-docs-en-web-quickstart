////////////////////////////////////////////////////////////////////////////////
// Отчёт "Гант по заказам на производство" (план/факт производства)
// Конфигурация: 1С:ERP Управление предприятием 2.5
//
// Структура строк (по образцу "ПРИМЕР.xlsx"):
//   Документ-основание (заказ клиента / давальца)
//     └ Заказ на производство      — интервал "План"
//         └ Этап производства      — интервал "Факт"
//
//   ПЛАН — от "Начать не ранее" заказа на производство,
//          длительностью "Срок изготовления, к дней"
//          (доп. реквизит документа-основания, календарные дни).
//          Ориентир окончания — "Желаемая дата выпуска" заказа на производство.
//   ФАКТ — по каждому этапу производства отдельной строкой:
//          фактическое начало/окончание этапа, а если факт ещё не проставлен —
//          плановые даты этапа.
//
// Параметры:
//   &СвойствоСрокИзготовления — ПланВидовХарактеристик.ДополнительныеРеквизитыИСведения
//                               (доп. реквизит "Срок изготовления, к дней")
//   &НачалоПериода, &КонецПериода — отбор по "Начать не ранее"; пустая дата = без ограничения
//   &Статусы — список статусов ЗНП (по умолчанию: Формируется, КПроизводству)
//
// ТОЧКИ АДАПТАЦИИ помечены [ПРОВЕРИТЬ] — см. docs/01-Анализ-ТЗ.md
////////////////////////////////////////////////////////////////////////////////

////////////////////////////////////////////////////////////////////////////////
// 1. Заказы на производство и их документы-основания
//    [ПРОВЕРИТЬ] Документ.ЗаказНаПроизводство2_5:
//       НачатьНеРанее        — "Начать не ранее"
//       ЖелаемаяДатаВыпуска  — "Желаемая дата выпуска"
//    [ПРОВЕРИТЬ] ТЧ Продукция: реквизит Заказ (заказ клиента / заказ давальца)
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ РАЗЛИЧНЫЕ
	ЗНП.Ссылка КАК Заказ,
	ЗНП.Номер КАК Номер,
	ЗНП.Дата КАК ДатаДокумента,
	ЗНП.Статус КАК Статус,
	ЗНП.Подразделение КАК Подразделение,
	ЗНП.НачатьНеРанее КАК НачатьНеРанее,
	ЗНП.ЖелаемаяДатаВыпуска КАК ЖелаемаяДатаВыпуска,
	Продукция.Заказ КАК ДокументОснование
ПОМЕСТИТЬ ВТ_Заказы
ИЗ
	Документ.ЗаказНаПроизводство2_5 КАК ЗНП
		ЛЕВОЕ СОЕДИНЕНИЕ Документ.ЗаказНаПроизводство2_5.Продукция КАК Продукция
		ПО Продукция.Ссылка = ЗНП.Ссылка
ГДЕ
	НЕ ЗНП.ПометкаУдаления
	И ЗНП.Статус В (&Статусы)
	И (&НачалоПериода = ДАТАВРЕМЯ(1, 1, 1) ИЛИ ЗНП.НачатьНеРанее >= &НачалоПериода)
	И (&КонецПериода = ДАТАВРЕМЯ(1, 1, 1) ИЛИ ЗНП.НачатьНеРанее <= &КонецПериода)

ИНДЕКСИРОВАТЬ ПО
	Заказ,
	ДокументОснование
;

////////////////////////////////////////////////////////////////////////////////
// 2. "Срок изготовления, к дней" из документа-основания.
//    Ищем и в табличной части ДополнительныеРеквизиты заказа клиента / давальца,
//    и в регистре ДополнительныеСведения — какой бы способ хранения ни был настроен.
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	З.Заказ КАК Заказ,
	З.ДокументОснование КАК ДокументОснование,
	ВЫРАЗИТЬ(ДопРеквКлиента.Значение КАК ЧИСЛО(10, 2)) КАК СрокИзготовленияДней
ПОМЕСТИТЬ ВТ_СрокВсеИсточники
ИЗ
	ВТ_Заказы КАК З
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Документ.ЗаказКлиента.ДополнительныеРеквизиты КАК ДопРеквКлиента
		ПО ДопРеквКлиента.Ссылка = З.ДокументОснование
			И ДопРеквКлиента.Свойство = &СвойствоСрокИзготовления

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	З.Заказ,
	З.ДокументОснование,
	ВЫРАЗИТЬ(ДопРеквДавальца.Значение КАК ЧИСЛО(10, 2))
ИЗ
	ВТ_Заказы КАК З
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Документ.ЗаказДавальца.ДополнительныеРеквизиты КАК ДопРеквДавальца
		ПО ДопРеквДавальца.Ссылка = З.ДокументОснование
			И ДопРеквДавальца.Свойство = &СвойствоСрокИзготовления

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	З.Заказ,
	З.ДокументОснование,
	ВЫРАЗИТЬ(ДопСведения.Значение КАК ЧИСЛО(10, 2))
ИЗ
	ВТ_Заказы КАК З
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ РегистрСведений.ДополнительныеСведения КАК ДопСведения
		ПО ДопСведения.Объект = З.ДокументОснование
			И ДопСведения.Свойство = &СвойствоСрокИзготовления
;

ВЫБРАТЬ
	СВ.Заказ КАК Заказ,
	СВ.ДокументОснование КАК ДокументОснование,
	МАКСИМУМ(СВ.СрокИзготовленияДней) КАК СрокИзготовленияДней
ПОМЕСТИТЬ ВТ_Срок
ИЗ
	ВТ_СрокВсеИсточники КАК СВ
ГДЕ
	СВ.СрокИзготовленияДней ЕСТЬ НЕ NULL

СГРУППИРОВАТЬ ПО
	СВ.Заказ,
	СВ.ДокументОснование

ИНДЕКСИРОВАТЬ ПО
	Заказ,
	ДокументОснование
;

////////////////////////////////////////////////////////////////////////////////
// 3. Этапы производства заказов — по строке на этап.
//    [ПРОВЕРИТЬ] Документ.ЭтапПроизводства2_5:
//       ЗаказНаПроизводство                      — ссылка на ЗНП
//       ДатаНачала / ДатаЗавершения              — план по графику
//       ДатаНачалаФакт / ДатаЗавершенияФакт      — факт ("Начат" / "Завершен")
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	Этапы.Ссылка КАК Этап,
	Этапы.Номер КАК НомерЭтапа,
	Этапы.ЗаказНаПроизводство КАК Заказ,
	Этапы.ДатаНачалаФакт КАК НачатФакт,
	Этапы.ДатаЗавершенияФакт КАК ЗавершенФакт,
	ВЫБОР
		КОГДА Этапы.ДатаНачалаФакт = ДАТАВРЕМЯ(1, 1, 1)
			ТОГДА Этапы.ДатаНачала
		ИНАЧЕ Этапы.ДатаНачалаФакт
	КОНЕЦ КАК ДатаНачалаИнтервала,
	ВЫБОР
		КОГДА Этапы.ДатаЗавершенияФакт = ДАТАВРЕМЯ(1, 1, 1)
			ТОГДА Этапы.ДатаЗавершения
		ИНАЧЕ Этапы.ДатаЗавершенияФакт
	КОНЕЦ КАК ДатаОкончанияИнтервала
ПОМЕСТИТЬ ВТ_Этапы
ИЗ
	Документ.ЭтапПроизводства2_5 КАК Этапы
ГДЕ
	НЕ Этапы.ПометкаУдаления
	И Этапы.ЗаказНаПроизводство В
			(ВЫБРАТЬ
				З.Заказ
			ИЗ
				ВТ_Заказы КАК З)

ИНДЕКСИРОВАТЬ ПО
	Заказ
;

////////////////////////////////////////////////////////////////////////////////
// 4. Результат.
//    Строка "План" — на уровне заказа на производство,
//    строки "Факт" — на уровне каждого этапа производства.
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	З.ДокументОснование КАК ДокументОснование,
	З.Заказ КАК Заказ,
	З.Номер КАК Номер,
	З.ДатаДокумента КАК ДатаДокумента,
	З.Статус КАК Статус,
	З.Подразделение КАК Подразделение,
	ЗНАЧЕНИЕ(Документ.ЭтапПроизводства2_5.ПустаяСсылка) КАК Этап,
	ВЫРАЗИТЬ("План" КАК СТРОКА(10)) КАК ВидИнтервала,
	1 КАК ПорядокВида,
	З.НачатьНеРанее КАК ДатаНачала,
	ВЫБОР
		КОГДА ЕСТЬNULL(С.СрокИзготовленияДней, 0) > 0
			ТОГДА ДОБАВИТЬКДАТЕ(З.НачатьНеРанее, ДЕНЬ, ВЫРАЗИТЬ(С.СрокИзготовленияДней КАК ЧИСЛО(10, 0)))
		КОГДА З.ЖелаемаяДатаВыпуска > З.НачатьНеРанее
			ТОГДА З.ЖелаемаяДатаВыпуска
		ИНАЧЕ ДОБАВИТЬКДАТЕ(З.НачатьНеРанее, ДЕНЬ, 1)
	КОНЕЦ КАК ДатаОкончания,
	З.НачатьНеРанее КАК НачатьНеРанее,
	З.ЖелаемаяДатаВыпуска КАК ЖелаемаяДатаВыпуска,
	ЕСТЬNULL(С.СрокИзготовленияДней, 0) КАК СрокИзготовленияДней,
	ДАТАВРЕМЯ(1, 1, 1) КАК НачатФакт,
	ДАТАВРЕМЯ(1, 1, 1) КАК ЗавершенФакт,
	ЛОЖЬ КАК ЭтоФакт
ИЗ
	ВТ_Заказы КАК З
		ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Срок КАК С
		ПО С.Заказ = З.Заказ
			И С.ДокументОснование = З.ДокументОснование
ГДЕ
	З.НачатьНеРанее <> ДАТАВРЕМЯ(1, 1, 1)

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	З.ДокументОснование,
	З.Заказ,
	З.Номер,
	З.ДатаДокумента,
	З.Статус,
	З.Подразделение,
	Э.Этап,
	ВЫРАЗИТЬ("Факт" КАК СТРОКА(10)),
	2,
	Э.ДатаНачалаИнтервала,
	ВЫБОР
		КОГДА Э.ДатаОкончанияИнтервала > Э.ДатаНачалаИнтервала
			ТОГДА Э.ДатаОкончанияИнтервала
		ИНАЧЕ ДОБАВИТЬКДАТЕ(Э.ДатаНачалаИнтервала, ДЕНЬ, 1)
	КОНЕЦ,
	З.НачатьНеРанее,
	З.ЖелаемаяДатаВыпуска,
	ЕСТЬNULL(С.СрокИзготовленияДней, 0),
	Э.НачатФакт,
	Э.ЗавершенФакт,
	ИСТИНА
ИЗ
	ВТ_Заказы КАК З
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ ВТ_Этапы КАК Э
		ПО Э.Заказ = З.Заказ
		ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Срок КАК С
		ПО С.Заказ = З.Заказ
			И С.ДокументОснование = З.ДокументОснование
ГДЕ
	Э.ДатаНачалаИнтервала <> ДАТАВРЕМЯ(1, 1, 1)

УПОРЯДОЧИТЬ ПО
	Статус,
	ДокументОснование,
	ДатаНачала,
	Номер,
	ПорядокВида
