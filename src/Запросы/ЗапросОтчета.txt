////////////////////////////////////////////////////////////////////////////////
// Отчёт "Гант по заказам на производство" (план/факт производства)
// Конфигурация: 1С:ERP Управление предприятием 2.5
//
// Логика (согласовано с постановщиком):
//   ПЛАН  — начало = дата начала заказа на производство,
//           окончание = начало + "Количество дней" (доп. реквизит документа-основания:
//           заказ клиента / заказ давальца).
//   ФАКТ  — только для заказов в статусе "К производству", по этапам производства:
//           min(факт начала, иначе план начала) ... max(факт завершения, иначе план завершения).
//
// Параметры:
//   &СвойствоКоличествоДней — ПланВидовХарактеристик.ДополнительныеРеквизитыИСведения
//                             (доп. реквизит "Количество дней")
//   &НачалоПериода, &КонецПериода — отбор по дате начала ЗНП; пустая дата = без ограничения
//   &Статусы — массив статусов ЗНП (по умолчанию: Формируется, КПроизводству)
//
// ТОЧКИ АДАПТАЦИИ помечены комментарием [ПРОВЕРИТЬ] — см. docs/01-Анализ-ТЗ.md
////////////////////////////////////////////////////////////////////////////////

////////////////////////////////////////////////////////////////////////////////
// 1. Заказы на производство + документы-основания
//    [ПРОВЕРИТЬ] Документ.ЗаказНаПроизводство2_5: реквизиты ДатаНачала / ДатаЗавершения
//    [ПРОВЕРИТЬ] ТЧ Продукция: реквизит Заказ (ссылка на заказ клиента / давальца)
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ РАЗЛИЧНЫЕ
	ЗНП.Ссылка КАК Заказ,
	ЗНП.Номер КАК Номер,
	ЗНП.Дата КАК ДатаДокумента,
	ЗНП.Статус КАК Статус,
	ЗНП.Подразделение КАК Подразделение,
	ЗНП.ДатаНачала КАК ДатаНачалаЗаказа,
	ЗНП.ДатаЗавершения КАК ДатаЗавершенияЗаказа,
	Продукция.Заказ КАК ДокументОснование,
	ПРЕДСТАВЛЕНИЕ(Продукция.Заказ) КАК ДокументОснованиеПредставление
ПОМЕСТИТЬ ВТ_ЗаказыОснования
ИЗ
	Документ.ЗаказНаПроизводство2_5 КАК ЗНП
		ЛЕВОЕ СОЕДИНЕНИЕ Документ.ЗаказНаПроизводство2_5.Продукция КАК Продукция
		ПО Продукция.Ссылка = ЗНП.Ссылка
ГДЕ
	НЕ ЗНП.ПометкаУдаления
	И ЗНП.Статус В (&Статусы)
	И (&НачалоПериода = ДАТАВРЕМЯ(1, 1, 1) ИЛИ ЗНП.ДатаНачала >= &НачалоПериода)
	И (&КонецПериода = ДАТАВРЕМЯ(1, 1, 1) ИЛИ ЗНП.ДатаНачала <= &КонецПериода)

ИНДЕКСИРОВАТЬ ПО
	Заказ,
	ДокументОснование
;

////////////////////////////////////////////////////////////////////////////////
// 2. Шапка заказа (одна строка на заказ)
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	ЗО.Заказ КАК Заказ,
	ЗО.Номер КАК Номер,
	ЗО.ДатаДокумента КАК ДатаДокумента,
	ЗО.Статус КАК Статус,
	ЗО.Подразделение КАК Подразделение,
	ЗО.ДатаНачалаЗаказа КАК ДатаНачалаЗаказа,
	ЗО.ДатаЗавершенияЗаказа КАК ДатаЗавершенияЗаказа,
	МАКСИМУМ(ЗО.ДокументОснованиеПредставление) КАК ДокументОснованиеПредставление,
	КОЛИЧЕСТВО(РАЗЛИЧНЫЕ ЗО.ДокументОснование) КАК КоличествоОснований
ПОМЕСТИТЬ ВТ_ЗаказыШапка
ИЗ
	ВТ_ЗаказыОснования КАК ЗО

СГРУППИРОВАТЬ ПО
	ЗО.Заказ,
	ЗО.Номер,
	ЗО.ДатаДокумента,
	ЗО.Статус,
	ЗО.Подразделение,
	ЗО.ДатаНачалаЗаказа,
	ЗО.ДатаЗавершенияЗаказа

ИНДЕКСИРОВАТЬ ПО
	Заказ
;

////////////////////////////////////////////////////////////////////////////////
// 3. "Количество дней" из документа-основания.
//    Ищем и в табличной части ДополнительныеРеквизиты, и в регистре
//    ДополнительныеСведения — какой бы вариант хранения ни был настроен.
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	ЗО.Заказ КАК Заказ,
	ВЫРАЗИТЬ(ДопРеквКлиента.Значение КАК ЧИСЛО(10, 2)) КАК КоличествоДней
ПОМЕСТИТЬ ВТ_ДниВсеИсточники
ИЗ
	ВТ_ЗаказыОснования КАК ЗО
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Документ.ЗаказКлиента.ДополнительныеРеквизиты КАК ДопРеквКлиента
		ПО ДопРеквКлиента.Ссылка = ЗО.ДокументОснование
			И ДопРеквКлиента.Свойство = &СвойствоКоличествоДней

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	ЗО.Заказ,
	ВЫРАЗИТЬ(ДопРеквДавальца.Значение КАК ЧИСЛО(10, 2))
ИЗ
	ВТ_ЗаказыОснования КАК ЗО
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Документ.ЗаказДавальца.ДополнительныеРеквизиты КАК ДопРеквДавальца
		ПО ДопРеквДавальца.Ссылка = ЗО.ДокументОснование
			И ДопРеквДавальца.Свойство = &СвойствоКоличествоДней

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	ЗО.Заказ,
	ВЫРАЗИТЬ(ДопСведения.Значение КАК ЧИСЛО(10, 2))
ИЗ
	ВТ_ЗаказыОснования КАК ЗО
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ РегистрСведений.ДополнительныеСведения КАК ДопСведения
		ПО ДопСведения.Объект = ЗО.ДокументОснование
			И ДопСведения.Свойство = &СвойствоКоличествоДней
;

ВЫБРАТЬ
	ДВ.Заказ КАК Заказ,
	МАКСИМУМ(ДВ.КоличествоДней) КАК КоличествоДней
ПОМЕСТИТЬ ВТ_Дни
ИЗ
	ВТ_ДниВсеИсточники КАК ДВ
ГДЕ
	ДВ.КоличествоДней ЕСТЬ НЕ NULL

СГРУППИРОВАТЬ ПО
	ДВ.Заказ

ИНДЕКСИРОВАТЬ ПО
	Заказ
;

////////////////////////////////////////////////////////////////////////////////
// 4. Факт по этапам производства.
//    [ПРОВЕРИТЬ] Документ.ЭтапПроизводства2_5: реквизиты ЗаказНаПроизводство,
//    ДатаНачала / ДатаЗавершения (план по графику),
//    ДатаНачалаФакт / ДатаЗавершенияФакт (факт).
//    Правило: берём факт, если он заполнен, иначе плановую дату этапа.
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	Этапы.ЗаказНаПроизводство КАК Заказ,
	МИНИМУМ(ВЫБОР
			КОГДА Этапы.ДатаНачалаФакт = ДАТАВРЕМЯ(1, 1, 1)
				ТОГДА Этапы.ДатаНачала
			ИНАЧЕ Этапы.ДатаНачалаФакт
		КОНЕЦ) КАК ДатаНачалаФакт,
	МАКСИМУМ(ВЫБОР
			КОГДА Этапы.ДатаЗавершенияФакт = ДАТАВРЕМЯ(1, 1, 1)
				ТОГДА Этапы.ДатаЗавершения
			ИНАЧЕ Этапы.ДатаЗавершенияФакт
		КОНЕЦ) КАК ДатаЗавершенияФакт,
	КОЛИЧЕСТВО(РАЗЛИЧНЫЕ Этапы.Ссылка) КАК КоличествоЭтапов,
	СУММА(ВЫБОР
			КОГДА Этапы.ДатаЗавершенияФакт = ДАТАВРЕМЯ(1, 1, 1)
				ТОГДА 0
			ИНАЧЕ 1
		КОНЕЦ) КАК ЗавершеноЭтапов
ПОМЕСТИТЬ ВТ_Факт
ИЗ
	Документ.ЭтапПроизводства2_5 КАК Этапы
ГДЕ
	НЕ Этапы.ПометкаУдаления
	И Этапы.ЗаказНаПроизводство В
			(ВЫБРАТЬ
				ЗШ.Заказ
			ИЗ
				ВТ_ЗаказыШапка КАК ЗШ)

СГРУППИРОВАТЬ ПО
	Этапы.ЗаказНаПроизводство

ИНДЕКСИРОВАТЬ ПО
	Заказ
;

////////////////////////////////////////////////////////////////////////////////
// 5. Результат: по две строки на заказ — "План" и "Факт".
//    Строка "Факт" выводится только там, где этапы уже сформированы.
////////////////////////////////////////////////////////////////////////////////
ВЫБРАТЬ
	ЗШ.Заказ КАК Заказ,
	ЗШ.Номер КАК Номер,
	ЗШ.ДатаДокумента КАК ДатаДокумента,
	ЗШ.Статус КАК Статус,
	ЗШ.Подразделение КАК Подразделение,
	ЗШ.ДокументОснованиеПредставление КАК ДокументОснование,
	ВЫРАЗИТЬ("План" КАК СТРОКА(10)) КАК ВидИнтервала,
	1 КАК ПорядокВида,
	ЗШ.ДатаНачалаЗаказа КАК ДатаНачала,
	ВЫБОР
		КОГДА ЕСТЬNULL(Д.КоличествоДней, 0) > 0
			ТОГДА ДОБАВИТЬКДАТЕ(ЗШ.ДатаНачалаЗаказа, ДЕНЬ, ВЫРАЗИТЬ(Д.КоличествоДней КАК ЧИСЛО(10, 0)))
		КОГДА ЗШ.ДатаЗавершенияЗаказа > ЗШ.ДатаНачалаЗаказа
			ТОГДА ЗШ.ДатаЗавершенияЗаказа
		ИНАЧЕ ДОБАВИТЬКДАТЕ(ЗШ.ДатаНачалаЗаказа, ДЕНЬ, 1)
	КОНЕЦ КАК ДатаОкончания,
	ЕСТЬNULL(Д.КоличествоДней, 0) КАК КоличествоДнейПлан,
	ЕСТЬNULL(Ф.КоличествоЭтапов, 0) КАК КоличествоЭтапов,
	ЕСТЬNULL(Ф.ЗавершеноЭтапов, 0) КАК ЗавершеноЭтапов,
	ЗШ.ДатаЗавершенияЗаказа КАК ДатаЗавершенияЗаказа,
	ЛОЖЬ КАК ЭтоФакт
ИЗ
	ВТ_ЗаказыШапка КАК ЗШ
		ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Дни КАК Д
		ПО Д.Заказ = ЗШ.Заказ
		ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Факт КАК Ф
		ПО Ф.Заказ = ЗШ.Заказ
ГДЕ
	ЗШ.ДатаНачалаЗаказа <> ДАТАВРЕМЯ(1, 1, 1)

ОБЪЕДИНИТЬ ВСЕ

ВЫБРАТЬ
	ЗШ.Заказ,
	ЗШ.Номер,
	ЗШ.ДатаДокумента,
	ЗШ.Статус,
	ЗШ.Подразделение,
	ЗШ.ДокументОснованиеПредставление,
	ВЫРАЗИТЬ("Факт" КАК СТРОКА(10)),
	2,
	Ф.ДатаНачалаФакт,
	ВЫБОР
		КОГДА Ф.ДатаЗавершенияФакт > Ф.ДатаНачалаФакт
			ТОГДА Ф.ДатаЗавершенияФакт
		ИНАЧЕ ДОБАВИТЬКДАТЕ(Ф.ДатаНачалаФакт, ДЕНЬ, 1)
	КОНЕЦ,
	ЕСТЬNULL(Д.КоличествоДней, 0),
	Ф.КоличествоЭтапов,
	Ф.ЗавершеноЭтапов,
	ЗШ.ДатаЗавершенияЗаказа,
	ИСТИНА
ИЗ
	ВТ_ЗаказыШапка КАК ЗШ
		ВНУТРЕННЕЕ СОЕДИНЕНИЕ ВТ_Факт КАК Ф
		ПО Ф.Заказ = ЗШ.Заказ
		ЛЕВОЕ СОЕДИНЕНИЕ ВТ_Дни КАК Д
		ПО Д.Заказ = ЗШ.Заказ
ГДЕ
	Ф.ДатаНачалаФакт <> ДАТАВРЕМЯ(1, 1, 1)

УПОРЯДОЧИТЬ ПО
	Статус,
	ДатаНачала,
	Номер,
	ПорядокВида
